cmake_minimum_required(VERSION 3.20)
project(ChadFBTSteamVR LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT, matching build.ps1 (cl's default /MT) and the README's import claims.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(driver_chadfbt SHARED src/driver.cpp)
target_include_directories(driver_chadfbt PRIVATE third_party/openvr src)
target_link_libraries(driver_chadfbt PRIVATE ws2_32)
set_target_properties(driver_chadfbt PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/chadfbt/bin/win64")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(driver_chadfbt PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/chadfbt/bin/win64")
endforeach()
add_executable(chadfbt-anchors src/anchors.cpp)
target_include_directories(chadfbt-anchors PRIVATE third_party/openvr src)
target_link_libraries(chadfbt-anchors PRIVATE ws2_32)

add_custom_command(TARGET chadfbt-anchors POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/third_party/openvr/openvr_api.dll" "$<TARGET_FILE_DIR:chadfbt-anchors>/openvr_api.dll")
include(CTest)
if(BUILD_TESTING)
 add_executable(chadfbt-wire-tests tests/wire_tests.cpp)
 target_include_directories(chadfbt-wire-tests PRIVATE src)
 add_test(NAME chadfbt-wire-tests COMMAND chadfbt-wire-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/pose-validity.txt" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/anchor-v2.hex")
endif()
if(BUILD_TESTING)
 add_executable(chadfbt-receiver-tests tests/receiver_tests.cpp)
 target_include_directories(chadfbt-receiver-tests PRIVATE src third_party/openvr)
 add_test(NAME chadfbt-receiver-tests COMMAND chadfbt-receiver-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/target-v1.hex" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/target-v2.hex")
 add_executable(chadfbt-tracker-role-tests tests/tracker_role_tests.cpp)
 target_include_directories(chadfbt-tracker-role-tests PRIVATE src third_party/openvr)
 add_test(NAME chadfbt-tracker-role-tests COMMAND chadfbt-tracker-role-tests)
endif()
